//!Activity 2: Debug the Mistake
--> Find the mistake and correct the code so that clicking the button increases the stock quantity by one.

import { Component } from "react";
class Stock extends Component {
  state = {
    quantity: 10
  };

  addStock = () => {
    this.state.quantity = this.state.quantity + 1;
  };

  render() {
    return (
      <>
        <h2>Available Stock: {this.state.quantity}</h2>
        <button onClick={this.addStock}>
          Add Stock
        </button>
      </>
    );
  }
}

export default Stock;


Questions:
1. Why is directly modifying state incorrect?
2. Which React method should be used?
3. Write the corrected addStock method.
